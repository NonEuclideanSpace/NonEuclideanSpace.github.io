按层编号的完全二叉树，编号之间关系固定：

左孩子=2i,右孩子=2i+1,父亲=⌊i/2⌋

一直/2== 从结点向根走
我的做法：

#include <bits/stdc++.h>
using namespace std;
int main(){
    int x,y;cin>>x>>y;
    deque<int>a,b;
    while(x>1){
        a.push_front(x);
        x/=2;
    }
    a.push_front(1);
    while(y>1){
        b.push_front(y);
        y/=2;
    }
    b.push_front(1);
    bool flag=false;
    for(int i=0;i<min(a.size(),b.size());i++){
        if(a[i]==b[i]){
            continue;
        }else{
            cout<<a[i-1]<<endl;
            flag=true;
            break;
        }
    }
    if(flag==false){
        cout<<a[min(a.size(),b.size())-1];
    }
}

踩过的坑：较短的路径整条都是较长路径的前缀时（x 是 y 的祖先、反过来、或者 x == y），比较循环会一直 continue 到结束，什么都不输出。修正办法是在循环结束后补一个分支：这种情况下答案就是较短路径的最后一个元素。补上之后，开头特判 x==1||y==1 也就不需要了，它只是这类情况里的一个特例。
更短的写法：编号大的结点一定不会在编号小的结点上面，所以每次让较大的那个往上走一步，直到两者相等

while(x!=y){
	if(x>y)x/=2;
	else y/=2;
}


我的做法：

#include <bits/stdc++.h>
using namespace std;
struct Node{
    char val;
    Node* left;
    Node* right;
};
vector<Node*>tree;
string s;
int cnt=1;
void build(Node* node){
    if(s[cnt]=='.'){
        cnt++;
    }else{
        node->left=new Node{s[cnt],nullptr,nullptr};
        tree.push_back(node->left);
        cnt++;         
        build(node->left);
    }
    if(s[cnt]=='.'){
        cnt++;
    }else{
        node->right=new Node{s[cnt],nullptr,nullptr};
        tree.push_back(node->right);
        cnt++;
        build(node->right);
    }
}
void midtraverse(Node *node){
    if(node==nullptr)return;
    midtraverse(node->left);
    cout<<node->val;
    midtraverse(node->right);
}
void posttraverse(Node *node){
    if(node==nullptr)return;
    posttraverse(node->left);
    posttraverse(node->right);
    cout<<node->val;
}
int main(){
    cin>>s;
    if(s[0]=='.')return 0;
    Node* root=new Node{s[0],nullptr,nullptr};
    tree.push_back(root);
    build(root);
    midtraverse(root);cout<<endl;
    posttraverse(root);cout<<endl;
    return 0;
}
